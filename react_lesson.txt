React бол Facebook (Meta) компанийн 2013 онд нээлттэй эхийн болгосон JavaScript сан (library) юм. Хэрэглэгчийн интерфэйс (UI) бүтээхэд зориулагдсан бөгөөд component-based архитектур ашигладаг. Өнөөдөр дэлхийн хамгийн алдартай frontend framework/library-ийн нэг бөгөөд Facebook, Instagram, Netflix, Airbnb зэрэг том компаниуд ашигладаг.

=== ҮНДСЭН ОЙЛГОЛТ ===

1. React гэж юу вэ?

React нь UI бүтээх declarative, component-based JavaScript library юм. Virtual DOM ашиглан хуудсыг хурдан шинэчилдэг. SPA (Single Page Application) бүтээхэд тохиромжтой.

Суулгах:

npx create-react-app my-app
cd my-app
npm start

Эсвэл Vite ашиглан:

npm create vite@latest my-app -- --template react
cd my-app
npm install
npm run dev

2. JSX - JavaScript XML

JSX нь HTML-тэй төстэй синтакс бөгөөд JavaScript дотор UI бичих боломж олгодог.

function Мэндчилгээ() {
  const нэр = "Бат";
  return (
    <div>
      <h1>Сайн байна уу, {нэр}!</h1>
      <p>Өнөөдөр {new Date().toLocaleDateString()} байна.</p>
    </div>
  );
}

JSX дүрмүүд:
- Нэг root element буцаана (эсвэл Fragment <>...</> ашиглана)
- class биш className ашиглана
- JavaScript илэрхийлэлийг {} дотор бичнэ
- Бүх tag хаагдсан байх ёстой (<img /> гэх мэт)

6. useEffect - Хажуугийн үйлдэл

import { useEffect, useState } from 'react';

function ӨгөгдөлТатах() {
  const [data, setData] = useState(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    fetch("https://api.example.com/data")
      .then(res => res.json())
      .then(data => {
        setData(data);
        setLoading(false);
      });
  }, []);  // [] - зөвхөн нэг удаа ажиллана

  if (loading) return <p>Уншиж байна...</p>;
  return <div>{JSON.stringify(data)}</div>;
}

useEffect dependency array:
- [] - mount хийхэд нэг удаа
- [утга] - утга өөрчлөгдөхөд
- байхгүй бол - render бүрд

Cleanup function:

useEffect(() => {
  const interval = setInterval(() => console.log("tick"), 1000);
  return () => clearInterval(interval);  // unmount хийхэд цэвэрлэнэ
}, []);

8. Conditional Rendering

function Мэдэгдэл({ төрөл, текст }) {
  if (!текст) return null;

  return (
    <div className={`alert alert-${төрөл}`}>
      {төрөл === "error" && <span>⚠️</span>}
      {төрөл === "success" && <span>✅</span>}
      <p>{текст}</p>
    </div>
  );
}

function Жагсаалт({ items }) {
  return (
    <div>
      {items.length === 0 ? (
        <p>Хоосон байна</p>
      ) : (
        <ul>
          {items.map(item => (
            <li key={item.id}>{item.нэр}</li>
          ))}
        </ul>
      )}
    </div>
  );
}

9. useRef ба DOM хандалт

import { useRef } from 'react';

function ФокусInput() {
  const inputRef = useRef(null);

  const handleClick = () => {
    inputRef.current.focus();
  };

  return (
    <div>
      <input ref={inputRef} type="text" />
      <button onClick={handleClick}>Фокус хийх</button>
    </div>
  );
}

Өмнөх утга хадгалах:

const өмнөх = useRef();

  өмнөх.current = утга;
}, [утга]);

10. Custom Hook

function useLocalStorage(key, initialValue) {
  const [утга, setУтга] = useState(() => {
    const saved = localStorage.getItem(key);
    return saved ? JSON.parse(saved) : initialValue;
  });

  useEffect(() => {
    localStorage.setItem(key, JSON.stringify(утга));
  }, [key, утга]);

  return [утга, setУтга];
}

function useFetch(url) {
  const [data, setData] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  useEffect(() => {
    fetch(url)
      .then(res => res.json())
      .then(setData)
      .catch(setError)
      .finally(() => setLoading(false));
  }, [url]);

  return { data, loading, error };
}

11. Context API - Global State

import { createContext, useContext, useState } from 'react';

const ThemeContext = createContext();

function ThemeProvider({ children }) {
  const [theme, setTheme] = useState("light");
  const toggle = () => setTheme(t => t === "light" ? "dark" : "light");

  return (
    <ThemeContext.Provider value={{ theme, toggle }}>
      {children}
    </ThemeContext.Provider>
  );
}

function useTheme() {
  return useContext(ThemeContext);
}

function Товч() {
  const { theme, toggle } = useTheme();
  return <button onClick={toggle}>Theme: {theme}</button>;
}

13. useReducer - Нарийн төлөв удирдах

import { useReducer } from 'react';

const initialState = { тоо: 0, түүх: [] };

function reducer(state, action) {
  switch (action.type) {
    case "НЭМЭХ":
      return { тоо: state.тоо + 1, түүх: [...state.түүх, "нэмсэн"] };
    case "ХАСАХ":
      return { тоо: state.тоо - 1, түүх: [...state.түүх, "хассан"] };
    case "ТЭГЛЭХ":
      return initialState;
    default:
      return state;
  }
}

function Тоолуур() {
  const [state, dispatch] = useReducer(reducer, initialState);

  return (
    <div>
      <p>Тоо: {state.тоо}</p>
      <button onClick={() => dispatch({ type: "НЭМЭХ" })}>+</button>
      <button onClick={() => dispatch({ type: "ХАСАХ" })}>-</button>
      <button onClick={() => dispatch({ type: "ТЭГЛЭХ" })}>Reset</button>
      <p>Түүх: {state.түүх.join(", ")}</p>
    </div>
  );
}

14. Performance Optimization

import { useMemo, useCallback, memo } from 'react';

const Жагсаалт = memo(function Жагсаалт({ items, onItemClick }) {
  return (
    <ul>
      {items.map(item => (
        <li key={item.id} onClick={() => onItemClick(item.id)}>
          {item.нэр}
        </li>
      ))}
    </ul>
  );
});

function App() {
  const [query, setQuery] = useState("");
  const [items] = useState([...]);

  const filtered = useMemo(() =>
    items.filter(i => i.нэр.includes(query)),
    [items, query]
  );

  const handleClick = useCallback((id) => {
    console.log("Clicked:", id);
  }, []);

  return (
    <div>
      <input value={query} onChange={e => setQuery(e.target.value)} />
      <Жагсаалт items={filtered} onItemClick={handleClick} />
    </div>
  );
}

15. Error Boundary

class ErrorBoundary extends React.Component {
  state = { hasError: false, error: null };

  static getDerivedStateFromError(error) {
    return { hasError: true, error };
  }

  componentDidCatch(error, info) {
    console.error("Алдаа:", error, info);
  }

  render() {
    if (this.state.hasError) {
      return <h1>Алдаа гарлаа: {this.state.error.message}</h1>;
    }
    return this.props.children;
  }
}

<ErrorBoundary>
  <App />
</ErrorBoundary>
